var svgLayer=document.getElementById('svgLayer');

function getPos(id,side){
  var el=document.getElementById(id);
  var r=el.getBoundingClientRect();
  var canvas=document.getElementById('canvas').getBoundingClientRect();
  var y=r.top+r.height/2-canvas.top;
  if(side==='right')return{x:r.right-canvas.left,y:y};
  if(side==='left')return{x:r.left-canvas.left,y:y};
  return{x:r.left+r.width/2-canvas.left,y:y};
}

function createPathEl(fromId,fromSide,toId,toSide){
  var from=getPos(fromId,fromSide);
  var to=getPos(toId,toSide);
  var dx=to.x-from.x;
  var dy=to.y-from.y;
  var d;
  if(Math.abs(dx)>Math.abs(dy)*0.8){
    var cpx=dx*0.4;
    d='M'+from.x+','+from.y+' C'+(from.x+cpx)+','+from.y+' '+(to.x-cpx)+','+to.y+' '+to.x+','+to.y;
  } else {
    var cpy=dy*0.4;
    d='M'+from.x+','+from.y+' C'+from.x+','+(from.y+cpy)+' '+to.x+','+(to.y-cpy)+' '+to.x+','+to.y;
  }
  var path=document.createElementNS('http://www.w3.org/2000/svg','path');
  path.setAttribute('d',d);
  path.setAttribute('class','flow-path');
  svgLayer.appendChild(path);
  var dot=document.createElementNS('http://www.w3.org/2000/svg','circle');
  dot.setAttribute('r','5');
  dot.setAttribute('class','flow-dot');
  svgLayer.appendChild(dot);
  return{path:path,dot:dot};
}

function animateConnector(fromId,fromSide,toId,toSide,duration){
  duration=duration||900;
  return new Promise(function(resolve){
    var result=createPathEl(fromId,fromSide,toId,toSide);
    var path=result.path,dot=result.dot;
    path.classList.add('show','active');
    var pathLen=path.getTotalLength();
    var startT=performance.now();
    dot.style.opacity='1';
    function tick(now){
      var t=Math.min((now-startT)/duration,1);
      var ease=t<0.5?2*t*t:-1+(4-2*t)*t;
      var pt=path.getPointAtLength(ease*pathLen);
      dot.setAttribute('cx',pt.x);
      dot.setAttribute('cy',pt.y);
      dot.style.opacity=t<0.05?t/0.05:(t>0.9?(1-t)/0.1:'1');
      if(t<1)requestAnimationFrame(tick);
      else{dot.style.opacity='0';path.classList.remove('active');resolve();}
    }
    requestAnimationFrame(tick);
  });
}

function activate(id,desc){
  var el=document.getElementById(id);
  el.classList.remove('done','trigger');
  el.classList.add('active');
  if(desc)document.getElementById('statusText').textContent=desc;
}
function startWorking(id){
  var el=document.getElementById(id);
  el.classList.remove('active');
  el.classList.add('working');
}
function done(id){
  var el=document.getElementById(id);
  el.classList.remove('active','trigger','working');
  el.classList.add('done');
}
function trigger(id){
  document.getElementById(id).classList.add('trigger');
}
function wakeAgent(boxId){

}
function doneAgent(boxId){

}
